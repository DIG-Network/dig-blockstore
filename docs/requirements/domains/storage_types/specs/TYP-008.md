## Summary

BlockStoreConfig contains all tunable parameters for opening and operating a BlockStore, with sensible defaults for production use.

---

## Specification

```rust
use std::path::PathBuf;

/// Configuration for opening a BlockStore.
///
/// All fields have sensible defaults for production use.
/// Use `BlockStoreConfig::default()` and override specific fields as needed.
#[derive(Debug, Clone)]
pub struct BlockStoreConfig {
    // --- Storage path ---

    /// Path to the RocksDB database directory.
    pub path: PathBuf,

    // --- Cache settings ---

    /// Maximum number of blocks to keep in the in-memory block cache.
    /// Default: 1000
    pub block_cache_capacity: usize,

    /// Maximum number of headers to keep in the in-memory header cache.
    /// Default: 2000
    pub header_cache_capacity: usize,

    /// Number of shards for the block and header caches.
    /// Must be a power of two. Default: 16
    pub cache_shards: usize,

    /// Whether to warm caches with recent blocks on open.
    /// Default: true
    pub warm_cache_on_open: bool,

    /// Heights of canonical chain to warm when `warm_cache_on_open` is true (CAC-006 precursor).
    /// Default: 64
    pub warm_cache_depth: u64,

    // --- RocksDB tuning ---

    /// RocksDB write buffer size in bytes.
    /// Default: 67_108_864 (64 MB)
    pub write_buffer_size: usize,

    /// RocksDB block cache size in bytes (shared across column families).
    /// Default: 134_217_728 (128 MB)
    pub block_cache_size: usize,

    /// Maximum number of open file descriptors for RocksDB.
    /// Default: 1000
    pub max_open_files: i32,

    // --- BlobDB and compression ---

    /// Whether to enable BlobDB for CF_BLOCKS (large value separation).
    /// Default: true
    pub enable_blob_db: bool,

    /// Whether to compress blocks with zstd before storage.
    /// Default: true
    pub compress_blocks: bool,

    /// Zstd compression level (1-22, higher = better compression, slower).
    /// Default: 3
    pub compression_level: i32,

    /// Whether to use a trained zstd dictionary for compression.
    /// Default: true
    pub use_compression_dict: bool,

    // --- Write pipeline ---

    /// Maximum number of blocks to batch in the write pipeline before flushing.
    /// Default: 64
    pub write_pipeline_batch_size: usize,

    /// Maximum time in milliseconds to wait before flushing a partial batch.
    /// Default: 100
    pub write_pipeline_flush_ms: u64,

    /// Bounded async channel capacity for the write pipeline (BLK-008 precursor).
    /// Default: 256
    pub write_pipeline_channel_capacity: usize,

    /// Whether to use synchronous writes (fsync after each write).
    /// Default: false (async writes for performance)
    pub sync_writes: bool,

    /// Sequential readahead hint in bytes (BLK-006 precursor).
    /// Default: 2_097_152 (2 MiB)
    pub readahead_size: usize,

    // --- Pruning ---

    /// Whether to enable the compaction filter for automatic pruning.
    /// Default: false
    pub enable_compaction_pruning: bool,

    /// Minimum block height to retain. Blocks below this height may be pruned.
    /// None means no pruning. Default: None
    pub min_retained_height: Option<u64>,
}

impl Default for BlockStoreConfig {
    fn default() -> Self {
        Self {
            path: PathBuf::from("data/blockstore"),
            block_cache_capacity: 1000,
            header_cache_capacity: 2000,
            cache_shards: 16,
            warm_cache_on_open: true,
            warm_cache_depth: 64,
            write_buffer_size: 64 * 1024 * 1024,     // 64 MB
            block_cache_size: 128 * 1024 * 1024,      // 128 MB
            max_open_files: 1000,
            enable_blob_db: true,
            compress_blocks: true,
            compression_level: 3,
            use_compression_dict: true,
            write_pipeline_batch_size: 64,
            write_pipeline_flush_ms: 100,
            write_pipeline_channel_capacity: 256,
            sync_writes: false,
            readahead_size: 2_097_152,
            enable_compaction_pruning: false,
            min_retained_height: None,
        }
    }
}
```

## Implementation Notes

- Define in `src/config.rs`.
- Manual `Default` impl is used instead of `#[derive(Default)]` because `PathBuf::default()` produces an empty path, but we want `"data/blockstore"`.
- `cache_shards` should be validated as a power of two during `BlockStore::open()`.
- The test config (STR-005) overrides many of these defaults with smaller values.
